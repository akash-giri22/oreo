const { Contract } = require('fabric-contract-api'); class StudentContract extends Contract {
// Seed the ledger with sample students. async InitLedger(ctx) {
const students = [
{ id: 'S001', name: 'Asha Patil', age: 22, course: 'MSc Computer Science', grade: 'A' },
{ id: 'S002', name: 'Rohan Mehta', age: 23, course: 'MSc Computer Science', grade: 'B+' },
{ id: 'S003', name: 'Sara Khan', age: 22, course: 'MSc Data Science', grade: 'A-' },
];
for (const s of students) { await ctx.stub.putState(s.id,
Buffer.from(JSON.stringify(s)));
}
}
// Add a new student; fails if the ID already exists. async CreateStudent(ctx, id, name, age, course, grade) {
if (await this.StudentExists(ctx, id)) {
throw new Error(`Student ${id} already exists`);
}
const student = { id, name, age: parseInt(age, 10), course, grade };
await	ctx.stub.putState(id,
Buffer.from(JSON.stringify(student))); return JSON.stringify(student);
}
// Return the student stored under this ID. async ReadStudent(ctx, id) {
const data = await ctx.stub.getState(id); if (!data || data.length === 0) {
throw new Error(`Student ${id} does not exist`);
}
return data.toString();
}
// Overwrite an existing student record.
async UpdateStudent(ctx, id, name, age, course, grade) { if (!(await this.StudentExists(ctx, id))) {
throw new Error(`Student ${id} does not exist`);
}
const student = { id, name, age: parseInt(age, 10), course, grade };
await	ctx.stub.putState(id, Buffer.from(JSON.stringify(student)));
return JSON.stringify(student);
}
// Remove a student from the ledger state. async DeleteStudent(ctx, id) {
if (!(await this.StudentExists(ctx, id))) {
throw new Error(`Student ${id} does not exist`);
}
await ctx.stub.deleteState(id);
}
// True if a student with this ID is on the ledger. async StudentExists(ctx, id) {
const data = await ctx.stub.getState(id); return !!data && data.length > 0;
}
// Return every student on the ledger. async GetAllStudents(ctx) {
const results = [];
const iterator = await ctx.stub.getStateByRange('', ''); let r = await iterator.next();
while (!r.done) {
results.push(JSON.parse(r.value.value.toString('utf8'))); r = await iterator.next();
}
await iterator.close();
return JSON.stringify(results);
}
}
module.exports = StudentContract;

StudentExists	Checks whether an ID is already on the ledger (helper) CreateStudent	Adds a new student; fails if the ID exists
ReadStudent	Returns the student with a given ID; fails if missing UpdateStudent	Overwrites an existing student; fails if missing DeleteStudent	Removes a student from the world state GetAllStudents	Returns every student on the ledger
